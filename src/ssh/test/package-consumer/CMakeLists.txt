cmake_minimum_required(VERSION 3.22)
project(dssh_package_c_consumer LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(DSSH_PACKAGE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(deucessh CONFIG REQUIRED)
get_property(languages GLOBAL PROPERTY ENABLED_LANGUAGES)
if(NOT languages STREQUAL "C" OR CMAKE_CXX_COMPILER_LOADED)
	message(FATAL_ERROR "The C-only package consumer unexpectedly enabled another language")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/package-c-language.txt" CONTENT "${languages}\n")
enable_testing()
if(UNIX)
	find_package(Threads REQUIRED)
	include(CheckSymbolExists)
	set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
	check_symbol_exists(thrd_create "threads.h" DSSH_PACKAGE_APP_C11)
	unset(CMAKE_REQUIRED_LIBRARIES)
	set(app_threads Threads::Threads)
	if(NOT DSSH_PACKAGE_APP_C11)
		find_library(DSSH_PACKAGE_APP_THREADS NAMES stdthreads REQUIRED)
		list(APPEND app_threads "${DSSH_PACKAGE_APP_THREADS}")
	endif()
endif()
set(forms "")
foreach(form IN ITEMS static shared)
	if(NOT TARGET deucessh::deucessh_${form})
		continue()
	endif()
	list(APPEND forms "${form}")
	add_executable(package_query_${form}
		"${CMAKE_CURRENT_SOURCE_DIR}/consumer.c")
	target_link_libraries(package_query_${form} PRIVATE deucessh::deucessh_${form})
	add_test(NAME package_query_${form} COMMAND package_query_${form})
	if(UNIX)
		add_executable(package_event_${form}
			"${DSSH_PACKAGE_SOURCE}/test/test_event_peer.c"
			"${DSSH_PACKAGE_SOURCE}/test/mock_io.c")
		target_compile_definitions(package_event_${form} PRIVATE _DEFAULT_SOURCE)
		target_link_libraries(package_event_${form}
			PRIVATE deucessh::deucessh_${form} ${app_threads})
		add_test(NAME package_event_${form} COMMAND package_event_${form})
	endif()
	foreach(kind IN ITEMS query event)
		if(TARGET package_${kind}_${form})
			if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
				target_compile_options(package_${kind}_${form} PRIVATE
					-pedantic-errors -Wall -Wextra -Wconversion -Werror)
			elseif(MSVC)
				target_compile_options(package_${kind}_${form} PRIVATE /W4 /WX)
			endif()
		endif()
	endforeach()
endforeach()
if(NOT forms)
	message(FATAL_ERROR "The installed package supplies no DeuceSSH library target")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/package-c-forms.txt" CONTENT "${forms}\n")
